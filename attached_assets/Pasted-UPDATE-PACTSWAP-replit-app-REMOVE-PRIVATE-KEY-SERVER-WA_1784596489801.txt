// 1. VERIFY PAYMENT - Check if txHash is real on BSC
app.post('/api/verify-payment', async (req, res) => {
  const { txHash, expectedAmount, expectedAddress } = req.body;
  
  try {
    // Call BSCScan API to check transaction
    const bscScanUrl = `https://api.bscscan.com/api?module=proxy&action=eth_getTransactionReceipt&txhash=${txHash}`;
    const receipt = await axios.get(bscScanUrl);
    
    if(receipt.data.result && receipt.data.result.status === "0x1") {
      // TX is confirmed
      // TODO: Update DB order status to "paid"
      res.json({ success: true, message: "Payment verified" });
    } else {
      res.json({ success: false, message: "Transaction not confirmed" });
    }
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});
