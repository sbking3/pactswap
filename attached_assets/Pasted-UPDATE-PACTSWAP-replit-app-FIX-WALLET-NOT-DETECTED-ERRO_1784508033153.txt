=== STEP 3: REPLACE YOUR connectWallet() FUNCTION WITH THIS ===
let provider;
let web3;
let accounts;

async function connectWallet() {
  document.getElementById("connectButton").innerText = "Connecting...";

  try {
    // 1. If MetaMask/Trust extension exists
    if (window.ethereum) {
      provider = window.ethereum;
    }
    // 2. If NO extension, use WalletConnect for Chrome
    else {
      provider = new WalletConnectProvider.default({
        rpc: { 56: "https://bsc-dataseed.binance.org/" },
        chainId: 56,
        qrcodeModalOptions: { mobileLinks: ["metamask", "trust"] }
      });
      await provider.enable();
    }

    web3 = new Web3(provider);
    accounts = await web3.eth.getAccounts();

    // Force BSC Network
    await switchToBSC();

    // 3. UPDATE BUTTON + FORM
    document.getElementById("connectButton").innerText = accounts[0].substring(0,6) + "..." + accounts[0].substring(38);
    if(document.getElementById("walletAddress")) {
      document.getElementById("walletAddress").value = accounts[0];
    }

    loadUserData(); // refresh balances

    // 4. LISTENERS TO PREVENT STUCK LOADING
    provider.on("accountsChanged", () => window.location.reload());
    provider.on("chainChanged", () => window.location.reload());
    provider.on("disconnect", () => window.location.reload());

  } catch (error) {
    document.getElementById("connectButton").innerText = "Connect Wallet";
    alert("Connection cancelled");
    console.error(error);
  }
}

async function switchToBSC() {
  try {
    await provider.request({ method: 'wallet_switchEthereumChain', params: [{ chainId: '0x38' }] });
  } catch (switchError) {
    if (switchError.code === 4902) {
      await provider.request({ method: 'wallet_addEthereumChain', params: [{
        chainId: '0x38',
        chainName: 'BSC Mainnet',
        rpcUrls: ['https://bsc-dataseed.binance.org/'],
        nativeCurrency: {name: 'BNB', symbol: 'BNB', decimals: 18}
      }]})
    }
  }
}
