STEP 6: REPLACE /api/orders/create/route.js WITH:
import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { sendEmail } from '@/lib/mailer'
export async function POST(req) {
  const { offerId, buyerWallet } = await req.json()
  const order = await prisma.order.create({ data: { offerId, buyerWallet, status: 'PENDING' }})
  const offer = await prisma.offer.findUnique({ where: { id: offerId }, include: { user: true }})
  await prisma.notification.create({ data: { userId: offer.userId, message: `New USDT Buy Order: ${order.id}`, link: `/orders` }})
  if(offer.user.email) await sendEmail(offer.user.email, "New Order on PACTSWAP", `<h2>New order: ${offer.amount} USDT for ₦${offer.price}</h2><a href="https://pactswap.replit.app/orders">View</a>`)
  await prisma.offer.update({ where: { id: offerId }, data: { isActive: false }})
  return NextResponse.json(order)
}
