import { PrismaClient } from '@prisma/client' // or your db
const prisma = new PrismaClient()

export default async function handler(req, res) {
  if (req.method !== 'POST') return res.status(405).end()
  
  const { wallet } = req.body
  const ADMIN_WALLET = "0xF71C8E88aA5e433603FB76afCea62D02b9751338"
  
  if (!wallet || wallet.toLowerCase() !== ADMIN_WALLET.toLowerCase()) {
    return res.status(403).json({ error: 'Unauthorized' })
  }

  // Find all active offers with NO active orders
  const offersToDelete = await prisma.offer.findMany({
    where: { 
      isActive: true, 
      status: 'Active',
      orders: { none: { status: { in: [0, 1] } // 0=awaiting, 1=paid
    }
  })
  
  const offerIds = offersToDelete.map(o => o.id)
  
  // BULK UPDATE - THIS IS THE KEY PART
  const result = await prisma.offer.updateMany({
    where: { id: { in: offerIds } },
    data: { isActive: false, status: 'Deleted' }
  })
  
  await prisma.adminLog.create({
    data: { action: 'CLEAR_MARKETPLACE', wallet, deleted: result.count }
  })
  
  return res.status(200).json({ 
    deleted: result.count, 
    skipped: 4 - result.count, // 4 = your current active offers
    message: `${result.count} offers cleared` 
  })
}
